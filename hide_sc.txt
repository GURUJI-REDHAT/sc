q1
# a. Detect ICMP Echo Requests from external host to internal network
alert icmp $EXTERNAL_NET any -> $HOME_NET any (msg:"ICMP Echo Request Ping Detected"; itype:8; sid:1000001; rev:1;)

# b. Detect HTTP requests containing /admin in URI
alert tcp $EXTERNAL_NET any -> $HOME_NET $HTTP_PORTS (msg:"HTTP Admin URI Access Detected"; flow:to_server,established; content:"/admin"; http_uri; nocase; sid:1000002; rev:1;)

# c. Detect DNS queries for suspicious domains ending with .xyz
alert udp $HOME_NET any -> $EXTERNAL_NET 53 (msg:"Suspicious DNS Query for .xyz Domain"; content:".xyz"; nocase; sid:1000003; rev:1;)

# d. Detect HTTP requests generated by Nmap NSE scripts
alert tcp $EXTERNAL_NET any -> $HOME_NET $HTTP_PORTS (msg:"Nmap NSE HTTP Script Detected"; flow:to_server,established; content:"Nmap Scripting Engine"; http_user_agent; nocase; sid:1000004; rev:1;)

# e. Detect SQL injection attempts containing ' OR 1=1 in HTTP traffic
alert tcp $EXTERNAL_NET any -> $HOME_NET $HTTP_PORTS (msg:"SQL Injection Attempt OR 1=1 Detected"; flow:to_server,established; content:"' OR 1=1"; nocase; sid:1000005; rev:1;)


q5
a. Scan ports on 192.168.1.10 even if the ICMP is blocked
nmap -Pn 192.168.1.10

b. Scan an IPv6 address 2001:db8::1
nmap -6 2001:db8::1

c. Scan only the top 1000 common ports on a host 192.168.1.10
nmap --top-ports 1000 192.168.1.10

d. Detect the service versions running on open ports 192.168.1.10
nmap -sV 192.168.1.10

e. Determine if a firewall is filtering ports for 192.168.1.10
nmap -sA 192.168.1.10

[extra]
For scanning ports when ping/ICMP is blocked:
nmap -Pn -sS 192.168.1.10

For service version detection with no ping:
nmap -Pn -sV 192.168.1.10

For firewall filtering check with reason shown:
nmap -sA --reason 192.168.1.10
